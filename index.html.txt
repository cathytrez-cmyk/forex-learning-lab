  <script>
    let rate = 1.3700;
    
    function changePip(pips) {
      rate = Math.round((rate + pips * 0.0001) * 10000) / 10000;
      document.getElementById('pair-rate').innerText = rate.toFixed(4);
      document.getElementById('pair-explain').innerText = `1 USD = ${rate.toFixed(4)} CAD`;
    }

    function toggleELI13(id) {
      const box = document.getElementById(id);
      box.classList.toggle('active');
    }

    function updateCandle() {
      const open = parseFloat(document.getElementById('inp-open').value);
      const close = parseFloat(document.getElementById('inp-close').value);
      
      document.getElementById('lbl-open').innerText = open.toFixed(4);
      document.getElementById('lbl-close').innerText = close.toFixed(4);

      const box = document.getElementById('candle-box');
      const status = document.getElementById('candle-status');

      if (close >= open) {
        box.classList.remove('bearish');
        status.innerText = "BULLISH CANDLE (Buyers in control)";
        status.style.color = "var(--success)";
      } else {
        box.classList.add('bearish');
        status.innerText = "BEARISH CANDLE (Sellers in control)";
        status.style.color = "var(--danger)";
      }
    }

    function calcRisk() {
      const bal = parseFloat(document.getElementById('calc-bal').value) || 0;
      const riskPct = parseFloat(document.getElementById('calc-risk').value) || 0;
      const stopPips = parseFloat(document.getElementById('calc-stop').value) || 1;

      const dollarRisk = bal * (riskPct / 100);
      const lotSize = dollarRisk / (stopPips * 10);

      document.getElementById('res-dollar').innerText = `$${dollarRisk.toFixed(2)}`;
      document.getElementById('res-lot').innerText = `${Math.max(0.01, lotSize).toFixed(2)} Lots`;
    }
  </script>
